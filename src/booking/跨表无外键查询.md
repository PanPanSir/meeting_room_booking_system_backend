async find(pageNo: number, pageSize: number, username: string, meetingRoomName: string, meetingRoomPosition: string, bookingTimeRangeStart: number, bookingTimeRangeEnd: number ) {
    const skipCount = (pageNo - 1) * pageSize;

    const [bookings, totalCount] = await this.entityManager.findAndCount(Booking, {
      where: {
        user: {
          username: Like(`%${username}%`)
        },
        room: {
          name: Like(`%${meetingRoomName}%`),
          location: Like(`%${meetingRoomPosition}%`)
        },
        startTime: Between(new Date(bookingTimeRangeStart), new Date(bookingTimeRangeEnd))
      },
      relations: {
        user: true,
        room: true,
      },
      skip: skipCount,
      take: pageSize
    });

    return {
      bookings,
      totalCount
    }
}

# 这个查询代码里，因为用了外键，所以findAndCount的语句很好写，如果不用外键的话，查询语句要怎么写？
# 在不使用外键的情况下，需要通过手动编写 JOIN 实现关联查询。以下是改造后的 TypeORM 查询方案及与原版的对比：
async find(...) {
  const skipCount = (pageNo - 1) * pageSize;

  // 手动创建 QueryBuilder
  const queryBuilder = this.entityManager.createQueryBuilder(Booking, 'b')
    .leftJoinAndSelect(User, 'u', 'b.userId = u.id') // 手动关联用户表
    .leftJoinAndSelect(Room, 'r', 'b.roomId = r.id') // 手动关联会议室表
    .where('u.username LIKE :username', { username: `%${username}%` })
    .andWhere('r.name LIKE :meetingRoomName', { meetingRoomName: `%${meetingRoomName}%` })
    .andWhere('r.location LIKE :meetingRoomPosition', { meetingRoomPosition: `%${meetingRoomPosition}%` })
    .andWhere('b.startTime BETWEEN :start AND :end', {
      start: new Date(bookingTimeRangeStart),
      end: new Date(bookingTimeRangeEnd)
    })
    .skip(skipCount)
    .take(pageSize);

  // 分页数据与总数统计分离
  const [bookings, totalCount] = await Promise.all([
    queryBuilder.getMany(),
    queryBuilder.getCount()
  ]);

  return { bookings, totalCount };
}
